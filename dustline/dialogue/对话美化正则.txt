function init() {
  var KEY = 'csmg.say.v1';
  var OLD = 'csmg_avatars';
  var MV1 = 'mvu.avatarChoice.v1';
  var MV2 = 'mvu.customAvatars.v1';
  var EVT = 'csmg-say-sync';
  var MOODS = ['默认', '微笑', '开心', '害羞', '凝重', '生气', '悲伤', '受伤', '惊讶', '战斗', '厌恶', '喜欢', '迷糊'];
  var ALIAS = {沐子: ['沐子', '李沐子'], 李沐子: ['李沐子', '沐子'], 祁谣: ['祁谣', '祁瑶'], 祁瑶: ['祁瑶', '祁谣'], 敖烈: ['敖烈', '熬烈'], 熬烈: ['熬烈', '敖烈']};
  var box = document.querySelector('.csmg-say');
  var panel = document.querySelector('.csmg-say-box');
  var nameInput = document.querySelector('.csmg-say-box .name');

  var store = {layout: 'after', faces: {}};

  function ls() {
    try { return window.parent.localStorage; } catch (e) { return localStorage; }
  }
  function parentDoc() {
    try { return window.parent.document; } catch (e) { return document; }
  }
  function ok(u) { return typeof u === 'string' && /^(data:image\/|https?:\/\/|blob:)/i.test(u); }
  function map(k) {
    try {
      var x = JSON.parse(ls().getItem(k) || '{}');
      return x && typeof x === 'object' && !Array.isArray(x) ? x : {};
    } catch (e) { return {}; }
  }
  function names(n) {
    n = String(n || '').trim();
    var extra = ALIAS[n] || [];
    return [n].concat(extra).filter(function (x, i, arr) { return x && arr.indexOf(x) === i; });
  }
  function norm(t) {
    t = String(t || '').trim() || '默认';
    return MOODS.indexOf(t) >= 0 ? t : '默认';
  }
  function read() {
    var x = map(KEY);
    var faces = x.faces && typeof x.faces === 'object' ? x.faces : {};
    var out = {};
    Object.keys(faces).forEach(function (n) {
      var b = faces[n];
      var c = {};
      if (!b || typeof b !== 'object') return;
      MOODS.forEach(function (m) { if (ok(b[m])) c[m] = b[m]; });
      if (Object.keys(c).length) out[n] = c;
    });
    return {layout: x.layout === 'before' ? 'before' : 'after', faces: out};
  }
  function save() {
    try { ls().setItem(KEY, JSON.stringify(store)); emit(); return true; }
    catch (e) { tip(e && e.name === 'QuotaExceededError' ? '本机存储已满' : '无法保存'); return false; }
  }
  function emit() {
    try { parentDoc().dispatchEvent(new CustomEvent(EVT)); } catch (e) {}
  }
  function pick(n, t) {
    var i, a = names(n), b, rec, list, hit;
    t = norm(t);
    for (i = 0; i < a.length; i++) { b = store.faces[a[i]]; if (t !== '默认' && b && ok(b[t])) return b[t]; }
    for (i = 0; i < a.length; i++) { b = store.faces[a[i]]; if (b && ok(b['默认'])) return b['默认']; }
    rec = map(OLD);
    for (i = 0; i < a.length; i++) {
      b = rec[a[i]];
      if (!b || typeof b !== 'object') continue;
      if (t !== '默认' && b.emotions && ok(b.emotions[t])) return b.emotions[t];
      if (ok(b.avatar)) return b.avatar;
    }
    rec = map(MV1);
    for (i = 0; i < a.length; i++) if (ok(rec[a[i]])) return rec[a[i]];
    rec = map(MV2);
    for (i = 0; i < a.length; i++) {
      list = rec[a[i]];
      if (Array.isArray(list)) {
        hit = list.filter(function (v) { return v && ok(v.url); })[0];
        if (hit) return hit.url;
      }
    }
    return '';
  }
  function tip(t) { if (tipEl) tipEl.textContent = t || ''; }
  function who() { return String(nameInput.value || '').trim(); }
  function setFace(n, t, u) {
    n = String(n || '').trim();
    t = norm(t);
    if (!n) { tip('先填角色名'); return false; }
    var b = Object.assign({}, store.faces[n] || {});
    if (u) b[t] = u; else delete b[t];
    if (Object.keys(b).length) store.faces[n] = b; else delete store.faces[n];
    if (!save()) return false;
    paint();
    panelView();
    return true;
  }
  function paint() {
    if (!box) return;
    box.setAttribute('data-side', store.layout);
    var loc = box.querySelector('.csmg-say-local');
    var u = pick(box.getAttribute('data-char'), box.getAttribute('data-type'));
    if (!loc) return;
    if (u) {
      loc.style.backgroundImage = 'url(' + JSON.stringify(u) + ')';
      box.classList.add('is-local');
    } else {
      loc.style.backgroundImage = '';
      box.classList.remove('is-local');
    }
  }
  function panelView() {
    document.querySelectorAll('.csmg-say-box [data-side]').forEach(function (b) {
      b.classList.toggle('on', b.getAttribute('data-side') === store.layout);
    });
    var bag = store.faces[who()] || {};
    moodsEl.querySelectorAll('.mood').forEach(function (b) {
      var k = b.getAttribute('data-mood');
      b.classList.toggle('on', k === mood);
      b.classList.toggle('has', !!bag[k]);
    });
    var u = pick(who(), mood);
    prev.replaceChildren();
    if (u) {
      var im = document.createElement('img');
      im.alt = '';
      im.src = u;
      prev.appendChild(im);
    } else prev.textContent = (who() || '容').slice(0, 1);
  }
  function shrink(url) {
    return new Promise(function (resolve) {
      var im = new Image();
      im.onload = function () {
        var w0 = im.naturalWidth || im.width || 1;
        var h0 = im.naturalHeight || im.height || 1;
        var s = Math.min(1, 360 / Math.max(w0, h0));
        var c = document.createElement('canvas');
        c.width = Math.max(1, Math.round(w0 * s));
        c.height = Math.max(1, Math.round(h0 * s));
        var x = c.getContext('2d');
        if (!x) return resolve(url);
        x.imageSmoothingEnabled = true;
        x.drawImage(im, 0, 0, c.width, c.height);
        var out = '';
        try { out = c.toDataURL('image/webp', 0.84); } catch (e) {}
        if (!ok(out)) { try { out = c.toDataURL('image/jpeg', 0.84); } catch (e) {} }
        resolve(ok(out) ? out : url);
      };
      im.onerror = function () { resolve(url); };
      im.src = url;
    });
  }
  function readFile(file) {
    return new Promise(function (resolve, reject) {
      if (!file) return reject(new Error('没有文件'));
      if (file.type && file.type.indexOf('image/') !== 0) return reject(new Error('请选图片'));
      if (file.size > 8 * 1024 * 1024) return reject(new Error('图片请小于 8MB'));
      var r = new FileReader();
      r.onerror = function () { reject(new Error('读不了图')); };
      r.onload = function () { resolve(String(r.result || '')); };
      r.readAsDataURL(file);
    });
  }
  function setLayout(side) {
    store.layout = side === 'before' ? 'before' : 'after';
    save();
    if (box) {
      box.classList.remove('is-flip');
      void box.offsetWidth;
      box.classList.add('is-flip');
    }
    paint();
    panelView();
  }

    if (box && !nameInput.value) nameInput.value = String(box.getAttribute('data-char') || '').trim();
    if (box) mood = norm(box.getAttribute('data-type'));
    panel.hidden = false;

  }

    b.onclick = function () { mood = k; panelView(); };
    b.ondblclick = function () { fileInput.click(); };
    b.oncontextmenu = function (e) {
      e.preventDefault();

    };

  document.querySelector('.csmg-say-av').addEventListener('click', openPanel);
  document.querySelector('.csmg-say-close').onclick = function () { panel.hidden = true; };
  document.querySelectorAll('.csmg-say-box [data-side]').forEach(function (b) {
    b.onclick = function () { setLayout(b.getAttribute('data-side')); };
  });
  nameInput.addEventListener('input', panelView);
  fileInput.addEventListener('change', function () {
    var f = fileInput.files && fileInput.files[0];
    fileInput.value = '';
    if (!f) return;
    readFile(f).then(shrink).then(function (u) {
      if (setFace(who(), mood, u)) tip('已保存 ' + who() + ' · ' + mood);
    }).catch(function (e) { tip(e.message || '上传失败'); });
  });
  document.querySelector('.csmg-say-url').onclick = function () {
    var v = String(urlInput.value || '').trim();
    if (!v) return tip('先粘贴地址');
    if (/^data:image\//i.test(v)) {
      shrink(v).then(function (u) { if (setFace(who(), mood, u)) { urlInput.value = ''; tip('已保存'); } });
      return;
    }
    if (!/^https?:\/\//i.test(v)) return tip('要用 http(s) 地址');
    fetch(v, {mode: 'cors', cache: 'no-store'}).then(function (r) {
      if (!r.ok) throw new Error('读不到图');
      return r.blob();
    }).then(function (blob) {
      return readFile(new File([blob], 'a.png', {type: blob.type || 'image/png'}));
    }).then(shrink).then(function (u) {
      if (setFace(who(), mood, u)) { urlInput.value = ''; tip('已保存'); }
    }).catch(function () {
      var im = new Image();
      im.onload = function () { if (setFace(who(), mood, v)) { urlInput.value = ''; tip('已记下网址'); } };
      im.onerror = function () { tip('线上图读不到'); };
      im.src = v;
    });
  };
  document.querySelector('.csmg-say-clear').onclick = function () { if (setFace(who(), mood, '')) tip('已清除'); };
  document.querySelector('.csmg-say-forget').onclick = function () {
    var n = who();
    if (!n || !store.faces[n]) return tip('没有本机头像');
    delete store.faces[n];
    if (save()) { paint(); panelView(); tip('已删除 ' + n); }
  };

  function reload() { store = read(); paint(); if (!panel.hidden) panelView(); }
  try { parentDoc().addEventListener(EVT, reload); } catch (e) {}
  window.addEventListener('storage', function (e) {

  });

  store = read();
  if (box) nameInput.value = String(box.getAttribute('data-char') || '').trim();
  paint();
  console.info('[对话美化]', 'v5', store.layout);
